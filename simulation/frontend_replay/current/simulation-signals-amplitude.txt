# ═══════════════════════════════════════════════════════════════════════════════
#  HELPERS
# ═══════════════════════════════════════════════════════════════════════════════
def to_frac(x, max_den=10**9) -> Fraction:
    return Fraction(x).limit_denominator(max_den)

def is_below(value, threshold: Fraction) -> Tuple[Fraction, bool]:
    fr = to_frac(abs(float(value)))
    return fr, fr < threshold

def frac_str(fr: Fraction) -> str:
    return f"{fr.numerator}/{fr.denominator}"

def pct(fr: Fraction) -> str:
    return f"{float(fr)*100:.4f}%"

def hr(char='─', n=78): print(char * n)
def section(t): print("\n" + "═"*78 + f"\n  {t}\n" + "═"*78)
def subsection(t): print("\n" + "─"*78 + f"\n  {t}\n" + "─"*78)


# ═══════════════════════════════════════════════════════════════════════════════
#  I1 — Average Waiting Time
# ═══════════════════════════════════════════════════════════════════════════════
def eq_I1(cfg):
    if cfg['verbose']: print("\n[I1] Average Waiting Time — Eq. 13, p.45")
    T = float(cfg['T_sim'])
    t = np.linspace(0.0, T, 3000); N = len(t)
    v = np.maximum(5.0 - 3.0*np.sin(0.5*t) + 0.5*np.random.randn(N), 0.0)
    for frac in (0.15, 0.45, 0.80):
        s = int(N*frac); v[max(0, s-20):min(N, s+20)] = 0.0
    ind = (v < 0.1).astype(float)
    W_trapz   = float(trapezoid(ind, t))
    W_simpson = float(simpson(ind, x=t))
    N_mc = 50000
    mc   = ind[np.random.randint(0, N, N_mc)]
    W_mc = T * float(mc.mean())
    W_se = T * float(mc.std(ddof=1)) / math.sqrt(N_mc)
    fr_ts, ok_ts = is_below(abs(W_trapz - W_simpson)/max(abs(W_trapz),1e-12), TH['1pct'])
    fr_mc, ok_mc = is_below(abs(W_trapz - W_mc),
                            5*to_frac(W_se) + TH['1pct']*to_frac(W_trapz))
    return dict(name='I1', W_trapz=W_trapz, W_simpson=W_simpson, W_mc=W_mc, W_se=W_se,
                frac=max(fr_ts, fr_mc), frac_ts=fr_ts, frac_mc=fr_mc,
                consistent=ok_ts and ok_mc,
                summary=f"trap={W_trapz:.4f} simp={W_simpson:.4f} MC={W_mc:.4f}±{W_se:.4f}")


# ═══════════════════════════════════════════════════════════════════════════════
#  I2 — Average Queue Length
# ═══════════════════════════════════════════════════════════════════════════════
def eq_I2(cfg):
    if cfg['verbose']: print("\n[I2] Average Queue Length — Eq. 4.3, p.39")
    n, K = cfg['n_agents'], cfg['K_steps']
    tv = np.linspace(0.0, 1.0, K)
    c = (0.3 + 0.4*np.arange(n)/max(n-1,1))[:, None]
    A = (4.0 + 8.0*np.random.rand(n))[:, None]
    Q = np.maximum(A*np.exp(-((tv-c)/0.3)**2) + 0.5*np.random.randn(n, K), 0.0)
    Q_direct = float(Q.sum()/(n*K))
    Q_trapz  = float(np.mean([trapezoid(Q[i,:], tv) for i in range(n)]))
    fr, ok = is_below(abs(Q_direct - Q_trapz)/max(abs(Q_direct),1e-12), TH['2pct'])
    return dict(name='I2', Q_direct=Q_direct, Q_trapz=Q_trapz, frac=fr, consistent=ok,
                summary=f"direct={Q_direct:.4f} trapz={Q_trapz:.4f} rel={pct(fr)}")


# ═══════════════════════════════════════════════════════════════════════════════
#  I3 — Average Travel Time
# ═══════════════════════════════════════════════════════════════════════════════
def eq_I3(cfg):
    if cfg['verbose']: print("\n[I3] Average Travel Time — Eq. 4.2, p.39")
    N = 1000
    t_arr = 10.0*np.random.rand(N)
    truth = np.maximum(60.0 + 15.0*np.random.randn(N), 10.0)
    t_dep = t_arr + truth
    T_avg = float((t_dep - t_arr).mean())
    T_ci  = 1.96*float((t_dep - t_arr).std(ddof=1))/math.sqrt(N)
    fr, ok = is_below(abs(T_avg - truth.mean()), to_frac(T_ci))
    return dict(name='I3', T_avg=T_avg, T_ci95=T_ci, T_truth=float(truth.mean()),
                frac=fr, consistent=ok,
                summary=f"T̄={T_avg:.4f}s 95%CI±{T_ci:.4f}s truth={truth.mean():.4f}s")


# ═══════════════════════════════════════════════════════════════════════════════
#  I4 — Throughput
# ═══════════════════════════════════════════════════════════════════════════════
def eq_I4(cfg):
    if cfg['verbose']: print("\n[I4] Throughput — Eq. 14, p.45")
    N, p = 800, 0.85
    T_sim = float(cfg['T_sim'])*20
    V = np.random.binomial(N, p)
    Thr = V/T_sim*3600.0
    batch = np.random.binomial(N, p, size=200)/T_sim*3600.0
    z = float((Thr - batch.mean())/(batch.std(ddof=1)+1e-12))
    fr, ok = is_below(abs(z), TH['3sigma'])
    return dict(name='I4', Thr=float(Thr), Thr_mean=float(batch.mean()), z=z,
                frac=fr, consistent=ok,
                summary=f"Thr={Thr:.1f} mean={batch.mean():.1f} z={z:.3f}σ")


# ═══════════════════════════════════════════════════════════════════════════════
#  I5 — AoI Decay Kernel
# ═══════════════════════════════════════════════════════════════════════════════
def eq_I5(cfg):
    if cfg['verbose']: print("\n[I5] AoI Decay Kernel — Eq. 3.11, p.32")
    lam = float(cfg['lambda_decay'])
    tau = np.linspace(0.0, 1000.0, 20001)
    kap = np.exp(-lam*tau)
    I_num = float(trapezoid(kap, tau)); I_ana = 1.0/lam
    fr, ok = is_below(abs(I_num-I_ana)/I_ana, TH['1pct'])
    return dict(name='I5', I_num=I_num, I_ana=I_ana, frac=fr, consistent=ok,
                summary=f"∫κ={I_num:.4f} 1/λ={I_ana:.4f} rel={pct(fr)}")


# ═══════════════════════════════════════════════════════════════════════════════
#  I6 — Vehicle Kinematics (Euler)
# ═══════════════════════════════════════════════════════════════════════════════
def eq_I6(cfg):
    if cfg['verbose']: print("\n[I6] Vehicle Kinematics — Forward Euler, p.22")
    dt = float(cfg['dt']); N = int(float(cfg['T_sim'])/dt)
    t = np.arange(N)*dt
    x = np.zeros(N); v = np.zeros(N); v[0] = 15.0
    for k in range(N-1):
        a = 0.3*math.sin(0.1*t[k]) - 0.1*(v[k] - 15.0)
        x[k+1] = x[k] + v[k]*dt
        v[k+1] = v[k] + a*dt
    xe = np.arange(N)*(15.0*dt)
    err = float(np.max(np.abs(xe - 15.0*t)))
    fr, ok = is_below(err, TH['1e-9'])
    return dict(name='I6', x_final=float(x[-1]), v_final=float(v[-1]),
                euler_err=err, frac=fr, consistent=ok,
                summary=f"x_final={x[-1]:.1f}m v_final={v[-1]:.3f}m/s err={err:.2e}")


# ═══════════════════════════════════════════════════════════════════════════════
#  I7 — MYCC Control Law
# ═══════════════════════════════════════════════════════════════════════════════
def eq_I7(cfg):
    if cfg['verbose']: print("\n[I7] MYCC Control Law — Eq. B.1, p.41")
    kd, ks, L_truck, target_gap = 0.2, 0.4, 10.0, 25.0
    dt = float(cfg['dt']); Nt, steps = 5, 300
    x = np.array([-i*(L_truck+28.0) for i in range(Nt)], dtype=float)
    v = np.full(Nt, 20.0)
    for _ in range(steps):
        u = np.zeros(Nt)
        gap_err = (x[:-1] - x[1:] - L_truck) - target_gap
        spd_err = v[:-1] - v[1:]
        u[1:] = kd*gap_err + ks*spd_err
        v += u*dt
        x += v*dt
    gaps = (x[:-1] - x[1:]) - L_truck
    err_max = float(np.max(np.abs(gaps - target_gap)))
    fr = to_frac(err_max)
    return dict(name='I7', gaps=gaps.tolist(), err_max=err_max, frac=fr,
                consistent=fr < Fraction(5,1),
                summary=f"mean_gap={gaps.mean():.3f}m max_err={err_max:.4f}m")


# ═══════════════════════════════════════════════════════════════════════════════
#  P1 — Truncated Log-Normal Latency
# ═══════════════════════════════════════════════════════════════════════════════
def eq_P1(cfg):
    if cfg['verbose']: print("\n[P1] Truncated Log-Normal — Eq. 3.6, p.30")
    mu, sg = cfg['mu_tau'], cfg['sigma_tau']
    lo, hi = float(cfg['tau_min']), float(cfg['tau_max'])
    dist = stats.lognorm(s=sg, scale=math.exp(mu))
    Flo, Fhi = dist.cdf(lo), dist.cdf(hi)
    N = 50000
    u = Flo + (Fhi-Flo)*np.random.rand(N)
    tau = dist.ppf(u)
    pdf_trunc = lambda t: dist.pdf(t)/(Fhi-Flo)
    mean_int, _ = quad(lambda t: t*pdf_trunc(t), lo, hi, limit=200)
    fr, ok = is_below(abs(tau.mean()-mean_int)/mean_int, TH['5pct'])
    return dict(name='P1', tau=tau, mean_emp=float(tau.mean()),
                mean_int=float(mean_int), frac=fr, consistent=ok,
                summary=f"emp={tau.mean():.4f} int={mean_int:.4f} rel={pct(fr)}")


# ═══════════════════════════════════════════════════════════════════════════════
#  P2 — Gilbert-Elliott Loss
# ═══════════════════════════════════════════════════════════════════════════════
def eq_P2(cfg):
    if cfg['verbose']: print("\n[P2] Gilbert-Elliott Loss — Eq. 3.7, p.30")
    pG, pB = float(cfg['pG']), float(cfg['pB'])
    p, q   = float(cfg['p_trans']), float(cfg['q_trans'])
    T = 10000
    state = np.ones(T, dtype=int); loss = np.zeros(T)
    rs = np.random.rand(T); rl = np.random.rand(T)
    for t in range(1, T):
        prev = state[t-1]
        if prev == 1:
            state[t] = 2 if rs[t] < p else 1
            loss[t]  = rl[t] < pG
        else:
            state[t] = 1 if rs[t] < q else 2
            loss[t]  = rl[t] < pB
    pi_G = q/(p+q); pi_B = p/(p+q)
    emp_G    = float((state == 1).mean())
    emp_loss = float(loss.mean())
    ana_loss = pi_G*pG + pi_B*pB
    frG, okG = is_below(abs(emp_G - pi_G), TH['2pp'])
    frL, okL = is_below(abs(emp_loss - ana_loss), TH['2pp'])
    return dict(name='P2', state=state, loss=loss, pi_G=pi_G,
                emp_G=emp_G, emp_loss=emp_loss, ana_loss=ana_loss,
                frac=max(frG, frL), consistent=okG and okL,
                summary=f"π_G={pi_G:.3f}/{emp_G:.3f} loss={emp_loss:.4f}/{ana_loss:.4f}")


# ═══════════════════════════════════════════════════════════════════════════════
#  P3 — Bernoulli Loss

def eq_P3(cfg):
    if cfg['verbose']: print("\n[P3] Bernoulli Loss — p.34")
    p = float(cfg['p_loss']); N = 100000
    L = (np.random.rand(N) < p).astype(int)
    p_hat = float(L.mean()); se = math.sqrt(p_hat*(1-p_hat)/N)
    z = (p_hat - p)/se
    fr, ok = is_below(abs(z), TH['4sigma'])
    return dict(name='P3', p=p, p_hat=p_hat, z=float(z), frac=fr, consistent=ok,
                summary=f"p={p:.3f} p̂={p_hat:.4f} z={z:.3f}σ")


# ═══════════════════════════════════════════════════════════════════════════════
#  P4 — Severity Index
# ═══════════════════════════════════════════════════════════════════════════════
def eq_P4(cfg):
    if cfg['verbose']: print("\n[P4] Severity Index — Eq. 3.8, p.30")
    wt, wp = 0.5, 0.5
    tmin, tmax = float(cfg['tau_min']), float(cfg['tau_max'])
    sc = [('S1', 0.0, 0.00), ('S2', 100.0, 0.05),
          ('S3', 300.0, 0.15), ('S4', 600.0, 0.30)]
    rho = np.array([wt*(t-tmin)/(tmax-tmin) + wp*pl for _, t, pl in sc])
    mono = bool(np.all(np.diff(rho) > 0))
    rng_ok = bool(np.all((rho >= 0) & (rho <= 1)))
    fr = to_frac(abs(rho[-1] - rho[0]))
    return dict(name='P4', rho=rho.tolist(), scenarios=[s[0] for s in sc],
                frac=fr, consistent=mono and rng_ok,
                summary=f"ρ={np.round(rho,4)} monotone={mono} in[0,1]={rng_ok}")


# ═══════════════════════════════════════════════════════════════════════════════
#  P5 — Logistic Degradation
# ═══════════════════════════════════════════════════════════════════════════════
def eq_P5(cfg):
    if cfg['verbose']: print("\n[P5] Logistic Degradation — Eq. 3.5, p.29")
    L, k, x0 = cfg['L'], cfg['k'], cfg['x0']
    f = lambda x: L/(1.0 + math.exp(-k*(x - x0)))
    h = 1e-4
    d_num = (f(x0+h) - f(x0-h))/(2*h)
    d_ana = L*k*math.exp(0)/(1+math.exp(0))**2
    fr, ok = is_below(abs(d_num-d_ana)/abs(d_ana), TH['1e-6'])
    sanity = abs(f(x0) - L/2.0) < 1e-9
    return dict(name='P5', d_num=float(d_num), d_ana=float(d_ana), frac=fr,
                consistent=ok and sanity,
                summary=f"d_num={d_num:.10f} d_ana={d_ana:.10f} rel={float(fr):.2e}")

# ═══════════════════════════════════════════════════════════════════════════════
#  P8 — Cooperative Reward
# ═══════════════════════════════════════════════════════════════════════════════
def eq_P8(cfg):
    if cfg['verbose']: print("\n[P8] Cooperative Reward — Eq. 3.9, p.31")
    beta = float(cfg['beta']); n = cfg['n_agents']
    q = 4.0 + 8.0*np.random.rand(n)
    r_loc  = -q
    r_glob = -(1.0/n)*q.sum()
    r = beta*r_loc + (1-beta)*r_glob
    ok_mean = abs(r.mean() - (beta*r_loc.mean() + (1-beta)*r_glob)) < 1e-9
    r_at_beta1 = 1.0*r_loc + 0.0*r_glob
    ok_b1 = bool(np.allclose(r_at_beta1, r_loc, atol=1e-9))
    r_at_beta0 = 0.0*r_loc + 1.0*r_glob
    ok_b0 = bool(np.allclose(r_at_beta0, np.full(n, r_glob), atol=1e-9))
    return dict(name='P8', r_loc=r_loc.tolist(), r_glob=float(r_glob), r=r.tolist(),
                frac=to_frac(0.0), consistent=ok_mean and ok_b1 and ok_b0,
                summary=f"mean_r={r.mean():.4f} r_global={r_glob:.4f} "
                        f"β=1✓={ok_b1} β=0✓={ok_b0}")

# ═══════════════════════════════════════════════════════════════════════════════
#  TRAFFIC GROWTH MODEL DIAGNOSTICS
# ═══════════════════════════════════════════════════════════════════════════════
def build_growth_models(cfg):
    rng = np.random.default_rng(42)
    N = 980
    t = np.arange(N, dtype=float)
    L, k, x0 = cfg['L'], cfg['k'], cfg['x0']
    base = L/(1.0 + np.exp(-k*(t - x0)))
    osc  = 8*np.sin(t/50) + 4*np.sin(t/17) + 3*rng.standard_normal(N)
    y    = np.maximum(base + osc, 0.0)

    logistic = lambda x: L/(1.0 + np.exp(-k*(x - x0)))
    poly1 = np.poly1d(np.polyfit(t, y, 1))
    poly2 = np.poly1d(np.polyfit(t, y, 2))

    def expm(x, a, b, c): return a*np.exp(b*x) + c
    popt, _ = curve_fit(expm, t, y, p0=[300, 0.001, 0], maxfev=20000)
    exp_fn = lambda x: expm(x, *popt)

    return t, y, [
        ('Logistic (Mechanistic)', logistic),
        ('Polynomial (Deg 2)', poly2),
        ('Polynomial (Deg 1)', poly1),
        ('Exponential', exp_fn),
    ]


def plot_residuals_comparison(cfg):
    t, y, models = build_growth_models(cfg)
    fig, axes = plt.subplots(2, 2, figsize=(16, 10), sharex=True, sharey=True)
    axes = axes.flatten()
    colors = ['tab:blue', 'tab:orange', 'tab:green', 'tab:red']
    for idx, (name, fn) in enumerate(models):
        residuals = y - fn(t)
        ax = axes[idx]
        ax.scatter(t, residuals, s=6, color=colors[idx], alpha=0.6)
        ax.axhline(0, color='black', linestyle='--', linewidth=1.2)

def run_marl_decay_simulation(cfg, lambda_candidates):
    n_agents = cfg['n_agents']; steps = cfg['K_steps']
    p_loss = cfg['p_loss']
    mu, sg = cfg['mu_tau'], cfg['sigma_tau']
    dist_samples = np.random.lognormal(mean=mu, sigma=sg,
                                       size=(n_agents, steps))
    loss_mask = np.random.rand(n_agents, steps) < p_loss
    perf = {}
    for lam in lambda_candidates:
        rewards, aois = [], []
        for i in range(n_agents):
            cur_aoi = 0.0; tot_r = 0.0
            for s in range(steps):
                d = dist_samples[i, s]
                if loss_mask[i, s]: cur_aoi += d
                else:               cur_aoi = d
                kw = np.exp(-lam*cur_aoi)
                tot_r += -(1.0 - kw)*10.0
                aois.append(cur_aoi)
            rewards.append(tot_r)
        perf[lam] = dict(rewards=float(np.mean(rewards)),
                         mean_aoi=float(np.mean(aois)))
    return perf


def plot_marl_decay_tradeoff(cfg):
    lambdas = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]
    sim = run_marl_decay_simulation(cfg, lambdas)
    rewards = [sim[l]['rewards'] for l in lambdas]
    aois    = [sim[l]['mean_aoi'] for l in lambdas]

    fig, ax1 = plt.subplots(figsize=(10, 6))
    c1 = '#D32F2F'
    ax1.set_xlabel('λ Decay Rate', fontweight='bold')
    ax1.set_ylabel('Mean Cumulative Reward', color=c1, fontweight='bold')
    ax1.plot(lambdas, rewards, color=c1, marker='o', markersize=8,
             linewidth=2.5, label='Mean Reward')
    ax1.tick_params(axis='y', labelcolor=c1)
    ax1.grid(True, linestyle='--', alpha=0.5)
    for l, r in zip(lambdas, rewards):
        ax1.annotate(f"{r:.1f}", (l, r), textcoords="offset points",
                     xytext=(0, 10), ha='center', fontsize=9,
                     color=c1, fontweight='bold')
    ax2 = ax1.twinx(); c2 = '#1976D2'
    ax2.set_ylabel('Mean AoI (ms)', color=c2, fontweight='bold')
    ax2.plot(lambdas, aois, color=c2, marker='s', linestyle='--',
             linewidth=2, label='Mean AoI')
    ax2.tick_params(axis='y', labelcolor=c2); ax2.set_ylim(0, max(aois)*1.3)
    plt.title('MARL Simulation: Performance vs. λ Decay Strategies',
              fontsize=14, fontweight='bold', pad=15)
    plt.tight_layout()
    plt.savefig('marl_decay_strategy_tradeoff.png', dpi=120)
    plt.close(fig)

    print("\n--- MARL decay sweep ---")
    for lam in lambdas:
        print(f"λ={lam:<5}| reward={sim[lam]['rewards']:9.2f} | "
              f"AoI={sim[lam]['mean_aoi']:6.2f} ms")

def is_below_mat(v, tn, td):
    v = abs(float(v))
    ok = v < (tn / td)
    try:
        f = Fraction(v).limit_denominator(int(1e9)) if v > 0 else Fraction(0)
        n, d = f.numerator, f.denominator
    except Exception:
        n, d = v, 1
    return n, d, ok

def trapz(y, x):
    y = np.asarray(y, dtype=float); x = np.asarray(x, dtype=float)
    return float(np.sum(0.5 * (y[1:] + y[:-1]) * np.diff(x)))

def simpson_num(y, x):
    y = np.asarray(y, dtype=float); x = np.asarray(x, dtype=float)
    n = len(x)
    if n % 2 == 0:
        n -= 1; y = y[:n]; x = x[:n]
    h = (x[-1] - x[0]) / (n - 1)
    s = y[0] + y[-1] + 4*np.sum(y[1:-1:2]) + 2*np.sum(y[2:-1:2])
    return h/3 * s

def _sz(sz):
    return (int(sz),) if np.isscalar(sz) else tuple(int(s) for s in sz)

def logncdf_safe(x, mu, sg): return stats.lognorm.cdf(x, s=sg, scale=math.exp(mu))
def lognpdf_safe(x, mu, sg): return stats.lognorm.pdf(x, s=sg, scale=math.exp(mu))
def logninv_safe(p, mu, sg): return stats.lognorm.ppf(p, s=sg, scale=math.exp(mu))
def lognrnd_safe(mu, sg, sz):
    return stats.lognorm.rvs(s=sg, scale=math.exp(mu), size=_sz(sz))
def normcdf_safe(x, mu, sg): return stats.norm.cdf(x, loc=mu, scale=sg)
def normpdf_safe(x, mu, sg): return stats.norm.pdf(x, loc=mu, scale=sg)
def exprnd_safe(mu, sz): return np.random.exponential(mu, size=_sz(sz))
def binornd_safe(N, p, sz=None):
    return int(np.random.binomial(N, p)) if sz is None else np.random.binomial(N, p, size=_sz(sz))

def eq_I1(cfg):
    if cfg.verbose: print('\n[I1] Average Waiting Time')
    T = cfg.T_sim; N = 3000; t = np.linspace(0, T, N)
    v = np.maximum(5 - 3*np.sin(0.5*t) + 0.5*np.random.randn(N), 0)
    for f in [0.15, 0.45, 0.80]:
        s = round(N*f); a = max(0, s-20); b = min(N, s+20); v[a:b] = 0
    ind = (v < 0.1).astype(float)
    Wt = trapz(ind, t); Ws = simpson_num(ind, t)
    mc = ind[np.random.randint(0, N, 50000)]; Wm = T*np.mean(mc)
    Wse = T*np.std(mc, ddof=1)/np.sqrt(50000)
    n1, d1, o1 = is_below_mat(abs(Wt-Ws)/max(abs(Wt), 1e-12), 1, 100)
    tol_mc = 5*abs(Wse) + abs(Wt/100)
    n2, d2, o2 = is_below_mat(abs(Wt-Wm), round(tol_mc*1e9), 1e9)
    return dict(name='I1', W_trapz=Wt, W_simpson=Ws, W_mc=Wm, W_se=Wse,
                frac=max(n1/d1, n2/d2), consistent=o1 and o2,
                t=t, v=v, ind=ind,
                summary=f"trap={Wt:.4f} simp={Ws:.4f} MC={Wm:.4f}±{Wse:.4f}")

def eq_I2(cfg):
    if cfg.verbose: print('\n[I2] Average Queue Length')
    n = cfg.n_agents; K = cfg.K_steps; tv = np.linspace(0, 1, K)
    c = (0.3 + 0.4*np.arange(n)/max(n-1, 1)).reshape(-1, 1)
    A = (4 + 8*np.random.rand(n)).reshape(-1, 1)
    Q = np.maximum(A * np.exp(-((tv - c)/0.3)**2) + 0.5*np.random.randn(n, K), 0)
    Qd = Q.sum()/(n*K)
    Qt = np.mean([trapz(Q[i, :], tv) for i in range(n)])
    Qs = np.mean([simpson_num(Q[i, :], tv) for i in range(n)])
    n1, d1, o1 = is_below_mat(abs(Qd-Qt)/max(abs(Qd), 1e-12), 2, 100)
    n2, d2, o2 = is_below_mat(abs(Qd-Qs)/max(abs(Qd), 1e-12), 2, 100)
    return dict(name='I2', Q_direct=Qd, Q_trapz=Qt, Q_simp=Qs,
                frac=max(n1/d1, n2/d2), consistent=o1 and o2,
                Q=Q, tv=tv, summary=f"dir={Qd:.4f} trap={Qt:.4f}")

def eq_I7(cfg):
    if cfg.verbose: print('\n[I7] MYCC Control Law')
    kd = 0.2; ks = 0.4; Lt = 10; ds = 25; dt = cfg.dt; Nv = 5; steps = 300
    x = np.array([-(i)*(Lt+28) for i in range(Nv)], dtype=float)
    v = 20*np.ones(Nv); xh = np.zeros((steps+1, Nv)); vh = np.zeros((steps+1, Nv))
    xh[0, :] = x; vh[0, :] = v
    for s in range(steps):
        u = np.zeros(Nv)
        ge = (x[:-1] - x[1:] - Lt) - ds; se = v[:-1] - v[1:]
        u[1:] = kd*ge + ks*se; v = v + u*dt; x = x + v*dt
        xh[s+1, :] = x; vh[s+1, :] = v
    gaps = (x[:-1] - x[1:]) - Lt; emax = np.max(np.abs(gaps - ds))
    n1, d1, ok = is_below_mat(emax, 5, 1)
    return dict(name='I7', gaps=gaps, err_max=emax, frac=n1/d1,
                consistent=ok, x_hist=xh, v_hist=vh, dt=dt,
                summary=f"gap={np.mean(gaps):.3f} err={emax:.4f}")

def eq_P1(cfg):
    if cfg.verbose: print('\n[P1] Truncated Log-Normal')
    mu = cfg.mu_tau; sg = cfg.sigma_tau; lo = cfg.tau_min; hi = cfg.tau_max
    Flo = logncdf_safe(lo, mu, sg); Fhi = logncdf_safe(hi, mu, sg); Z = Fhi - Flo
    tau = logninv_safe(Flo + Z*np.random.rand(50000), mu, sg)
    mean_int, _ = quad(lambda t: t * lognpdf_safe(t, mu, sg)/Z, lo, hi)
    n1, d1, ok = is_below_mat(abs(np.mean(tau) - mean_int)/mean_int, 5, 100)
    return dict(name='P1', tau=tau, mean_emp=np.mean(tau), mean_int=mean_int,
                frac=n1/d1, consistent=ok, mu=mu, sg=sg, F_lo=Flo, F_hi=Fhi,
                summary=f"emp={np.mean(tau):.4f} int={mean_int:.4f}")

def plot_P1(cfg, r):

def eq_P2(cfg):
    if cfg.verbose: print('\n[P2] Gilbert-Elliott Loss')
    pG = cfg.pG; pB = cfg.pB; p = cfg.p_trans; q = cfg.q_trans
    T = 10000; st = np.ones(T, dtype=int); ls = np.zeros(T)
    rs = np.random.rand(T); rl = np.random.rand(T)
    for t in range(1, T):
        # Carry over previous state by default (otherwise state B resets to G!)
        st[t] = st[t-1]
        if st[t-1] == 1:
            if rs[t] < p: st[t] = 2
            ls[t] = float(rl[t] < pG)
        else:
            if rs[t] < q: st[t] = 1
            ls[t] = float(rl[t] < pB)
    piG = q/(p+q); piB = p/(p+q); eG = np.mean(st == 1); eL = np.mean(ls)
    aL = piG*pG + piB*pB
    n1, d1, o1 = is_below_mat(abs(eG - piG), 2, 100)
    n2, d2, o2 = is_below_mat(abs(eL - aL), 2, 100)
    return dict(name='P2', state=st, loss=ls, pi_G=piG, emp_G=eG,
                emp_loss=eL, ana_loss=aL, frac=max(n1/d1, n2/d2),
                consistent=o1 and o2,
                summary=f"piG={piG:.3f}/{eG:.3f} loss={eL:.4f}/{aL:.4f}")

def plot_P2(cfg, r):
    ns = 150; st = r['state'][:ns]; ls = r['loss'][:ns]; t = np.arange(1, ns+1)

    zero = ls == 0; one = ~zero
    axs[1].scatter(t[zero], np.zeros(np.sum(zero)), c='g', s=60, marker='.')

def eq_P4(cfg):

                frac=n1/d1, consistent=mono and inr,

def eq_P5(cfg):
    if cfg.verbose: print('\n[P5] Logistic Degradation')
    L = cfg.L; k = cfg.k; x0 = cfg.x0
    f = lambda x: L/(1 + math.exp(-k*(x - x0)))
    h = 1e-4; dn = (f(x0+h) - f(x0-h))/(2*h); da = L*k/4
    n1, d1, ok = is_below_mat(abs(dn - da)/abs(da), 1, 1e6)
    san = abs(f(x0) - L/2) < 1e-9
    return dict(name='P5', d_num=dn, d_ana=da, frac=n1/d1,
                consistent=ok and san, summary=f"dn={dn:.8f} da={da:.8f}")

def eq_A18(cfg):
    if cfg.verbose: print('\n[A18] M/M/1 Steady State')
    rho = 0.7; lam = rho; mu = 1.0
    pn_full = (1-rho)*rho**np.arange(21)
    # Renormalize pn to [0, 20] window to match the truncated empirical histogram
    pn = pn_full / pn_full.sum()
    # Simulate proper M/M/1: at each step RECORD the current state, THEN transition.
    # This correctly captures the empty-queue probability P(Q=0) = 1-ρ.
    N = 300000; st = 0; cnt = np.zeros(21); total_visits_in_range = 0
    p_arr = lam/(lam + mu)   # prob of arrival event (given an event occurs)
    for _ in range(N):
        # Record current state FIRST
        if 0 <= st <= 20:
            cnt[st] += 1
            total_visits_in_range += 1
        # Then transition
        if st == 0:
            st = 1   # empty queue: arrival must occur
        else:
            if np.random.rand() < p_arr: st += 1
            else: st -= 1
    emp = cnt / max(total_visits_in_range, 1)
    err = float(np.max(np.abs(emp - pn)))
    n1, d1, ok = is_below_mat(err, 5, 100)
    return dict(name='A18', p_n=pn, emp=emp, frac=n1/d1, consistent=ok, rho=rho,
                summary=f"rho={rho:.2f} err={err:.4f}")

def eq_B1(cfg):

def eq_B2(cfg):
    if cfg.verbose: print('\n[B2] Policy Improvement')
    n = 4; a = 2; np_state = np.random.get_state(); np.random.seed(7)
    P = np.zeros((n, a, n)); R = np.zeros((n, a))
    for s in range(n):
        for act in range(a):
            P[s, act, :] = np.random.rand(n); P[s, act, :] /= P[s, act, :].sum()
            R[s, act] = np.random.rand()*2 - 0.5
    np.random.set_state(np_state)
    g = 0.9; pol = np.zeros(n, dtype=int)
    Ppi = np.zeros((n, n)); Rpi = np.zeros(n)
    for s in range(n):
        Ppi[s, :] = P[s, pol[s], :]; Rpi[s] = R[s, pol[s]]
    Vold = np.linalg.solve(np.eye(n) - g*Ppi, Rpi)
    Qold = np.zeros((n, a))
    for s in range(n):
        for act in range(a):
            Qold[s, act] = R[s, act] + g*P[s, act, :]@Vold
    poln = np.argmax(Qold, axis=1)
    Ppi2 = np.zeros((n, n)); Rpi2 = np.zeros(n)
    for s in range(n):
        Ppi2[s, :] = P[s, poln[s], :]; Rpi2[s] = R[s, poln[s]]
    Vnew = np.linalg.solve(np.eye(n) - g*Ppi2, Rpi2)
    imp = Vnew - Vold; ok1 = bool(np.all(imp >= -1e-9))
    n1, d1, ok2 = is_below_mat(float(np.max(np.abs(np.min(imp)))), 1, 1e9)
    return dict(name='B2', V_old=Vold, V_new=Vnew, improvement=imp,

                summary=f"min(Vnew-Vold)={float(np.min(imp)):.4e}")

def eq_A31(cfg):
    g = 0.9; a = 3; Rrow = np.array([1.0, 2.0, 1.5])
    # Single-state MDP: Q*(s,a) = R(a) + γ·max_a' Q*(s,a') with
    # max_a' Q*(s,a') = max R / (1-γ), so Q_star = R + γ·max(R)/(1-γ)
    Q_max = float(np.max(Rrow)) / (1 - g)
    Q_star = Rrow + g * Q_max
    alpha = 0.05; T = 50000
    Qa = np.zeros(a); Qb = np.zeros(a); Qs = np.zeros(a)
    for _ in range(T):
        act = np.random.randint(0, a); rt = Rrow[act]
        if np.random.rand() < 0.5:
            aStar = int(np.argmax(Qa))
            Qa[act] += alpha*(rt + g*Qb[aStar] - Qa[act])
        else:
            aStar = int(np.argmax(Qb))
            Qb[act] += alpha*(rt + g*Qa[aStar] - Qb[act])
        aS = int(np.argmax(Qs))
        Qs[act] += alpha*(rt + g*Qs[aS] - Qs[act])
    Q_double = (Qa + Qb)/2
    err_d = float(np.max(np.abs(Q_double - Q_star)))
    err_s = float(np.max(np.abs(Qs - Q_star)))
    n1, d1, ok = is_below_mat(err_d, 5, 100)
    return dict(name='A31', Q_double=Q_double, Q_single=Qs, Q_star=Q_star,
                err_d=err_d, err_s=err_s, frac=n1/d1, consistent=ok,

def eq_A45(cfg):
    """Hazard rate identity: ∫_lo^hi h(x) dx = log(S(lo)/S(hi)) for non-truncated dist."""
    mu = cfg.mu_tau; sg = cfg.sigma_tau
    lo = max(cfg.tau_min, 1e-3); hi = cfg.tau_max
    # Use the non-truncated lognormal: S(x) = 1 - F(x)
    def S(x): return 1.0 - logncdf_safe(x, mu, sg)
    def h(x):
        f_pdf = lognpdf_safe(x, mu, sg)
        s = S(x)
        return f_pdf / max(s, 1e-12)
    H_int, _ = quad(h, lo, hi, limit=200)
    # Identity: ∫_lo^hi h(x) dx = log(S(lo)/S(hi))
    H_ana = math.log(max(S(lo), 1e-300) / max(S(hi), 1e-300))
    err = abs(H_int - H_ana) / max(abs(H_ana), 1.0)
    n1, d1, ok = is_below_mat(err, 1, 1e4)
    return dict(name='A45', H_int=H_int, H_ana=H_ana, frac=n1/d1, consistent=ok,
                summary=f"∫h dτ={H_int:.4f} vs log(S(lo)/S(hi))={H_ana:.4f}")

    err = float(np.max(np.abs(grad_num - grad_ana)))
    n1, d1, ok = is_below_mat(err, 1, 1e6)
    return dict(name='A73', err=err, frac=n1/d1, consistent=ok,
                summary=f"||∇(xᵀAx) num - ana||={err:.2e}")

def eq_A75(cfg):
    """RK4 vs Euler for dy/dt = -y (analytic: y(t) = y0·e^(-t))."""
    y0 = 1.0; T = 1.0; N = 100; dt = T/N
    t = np.linspace(0, T, N+1)
    y_euler = np.zeros(N+1); y_euler[0] = y0
    y_rk4 = np.zeros(N+1); y_rk4[0] = y0
    f = lambda y: -y
    for k in range(N):
        y_euler[k+1] = y_euler[k] + dt*f(y_euler[k])
        k1 = f(y_rk4[k]); k2 = f(y_rk4[k] + 0.5*dt*k1)
        k3 = f(y_rk4[k] + 0.5*dt*k2); k4 = f(y_rk4[k] + dt*k3)
        y_rk4[k+1] = y_rk4[k] + (dt/6)*(k1 + 2*k2 + 2*k3 + k4)
    y_ana = y0 * np.exp(-t)
    err_euler = float(np.max(np.abs(y_euler - y_ana)))
    err_rk4 = float(np.max(np.abs(y_rk4 - y_ana)))
    ok = err_rk4 < err_euler
    n1, d1, _ = is_below_mat(err_rk4, 1, 1e9)
    return dict(name='A75', err_euler=err_euler, err_rk4=err_rk4,
                frac=n1/d1, consistent=ok,
                summary=f"err_euler={err_euler:.2e} err_rk4={err_rk4:.2e}")

def eq_A85(cfg):
    """Sample variance unbiasedness: E[s²] = σ²."""
    N = 50000; n = 50; sigma_true = 2.0
    sample_vars = np.zeros(N)
    for trial in range(N):
        X = sigma_true * np.random.randn(n)
        sample_vars[trial] = np.var(X, ddof=1)
    mean_var = float(np.mean(sample_vars))
    err = abs(mean_var - sigma_true**2) / sigma_true**2
    n1, d1, ok = is_below_mat(err, 5, 100)
    return dict(name='A85', mean_var=mean_var, sigma2=sigma_true**2,
                frac=n1/d1, consistent=ok,
                summary=f"E[s²]={mean_var:.4f} σ²={sigma_true**2:.4f}")

def eq_A86(cfg):
    """Bessel's correction: E[s²_biased] = (n-1)/n · σ² < σ²."""
    N = 50000; n = 50; sigma_true = 2.0
    biased_vars = np.zeros(N)
    for trial in range(N):
        X = sigma_true * np.random.randn(n)
        biased_vars[trial] = np.var(X, ddof=0)   # biased (divide by n)
    mean_var = float(np.mean(biased_vars))
    expected_biased = (n - 1) / n * sigma_true**2
    err = abs(mean_var - expected_biased) / expected_biased
    n1, d1, ok = is_below_mat(err, 5, 100)
    return dict(name='A86', mean_var=mean_var, expected_biased=expected_biased,
                frac=n1/d1, consistent=ok,
                summary=f"E[s²_biased]={mean_var:.4f} (n-1)/n·σ²={expected_biased:.4f}")

def eq_A87(cfg):
    """Covariance vs independence: independent ⇒ Cov=0; converse false."""

def eq_A88(cfg):
    """Pearson correlation ρ ∈ [-1, 1]."""
    N = 100000
    X = np.random.randn(N)
    Y = 0.7*X + np.sqrt(1 - 0.49)*np.random.randn(N)
    muX, muY = X.mean(), Y.mean()
    num = np.mean((X - muX) * (Y - muY))
    den = np.std(X, ddof=0) * np.std(Y, ddof=0)
    rho = float(num / den)
    in_range = (-1.0 - 1e-9) <= rho <= (1.0 + 1e-9)
    err = abs(rho - 0.7)   # expected correlation
    n1, d1, ok = is_below_mat(err, 5, 100)
    return dict(name='A88', rho=rho, frac=n1/d1, consistent=ok and in_range,
                summary=f"ρ={rho:.4f} (expect 0.7)")

def eq_A89(cfg):
    """Spearman rank correlation = Pearson on ranks; monotonic ⇒ 1."""
    N = 10000
    X = np.random.randn(N)
    Y = X + 0.1*np.random.randn(N)   # near-monotonic
    from scipy.stats import spearmanr
    rho_s, _ = spearmanr(X, Y)
    rho_s = float(rho_s)
    in_range = (-1.0 - 1e-9) <= rho_s <= (1.0 + 1e-9)
    ok = in_range and rho_s > 0.95   # near-monotonic
    n1, d1, _ = is_below_mat(1 - rho_s, 5, 100)
    return dict(name='A89', rho_s=rho_s, frac=n1/d1, consistent=ok,
                summary=f"ρ_s={rho_s:.4f} (near-monotonic)")
